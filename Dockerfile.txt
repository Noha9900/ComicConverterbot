FROM python:3.11-slim

# Real 'unrar' is required to extract .cbr (RAR) archives, including RAR5.
# Debian's default 'unrar-free' package cannot read RAR5, so we fetch the
# official RARLab binary instead.
RUN apt-get update && \
    apt-get install -y --no-install-recommends wget ca-certificates && \
    wget -q https://www.rarlab.com/rar/rarlinux-x64-701.tar.gz -O /tmp/unrar.tar.gz && \
    tar -xzf /tmp/unrar.tar.gz -C /tmp && \
    cp /tmp/rar/unrar /usr/local/bin/unrar && \
    chmod +x /usr/local/bin/unrar && \
    rm -rf /tmp/* && \
    apt-get purge -y wget && \
    apt-get autoremove -y && \
    rm -rf /var/lib/apt/lists/*

WORKDIR /app

COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY . .

RUN mkdir -p downloads output data

ENV PYTHONUNBUFFERED=1

# No venv: dependencies are installed straight into the image's system
# Python, so `docker run` starts the bot immediately with nothing else
# to set up on the VPS.
CMD ["python", "bot.py"]
